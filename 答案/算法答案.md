# 算法答案

## LINUX-C-001 手写快速排序

```cpp
#include <vector>
#include <utility>

void quickSort(std::vector<int>& a, int l, int r) {
    if (l >= r) return;

    int pivot = a[l + (r - l) / 2];
    int i = l, j = r;

    while (i <= j) {
        while (a[i] < pivot) ++i;
        while (a[j] > pivot) --j;

        if (i <= j) {
            std::swap(a[i], a[j]);
            ++i;
            --j;
        }
    }

    if (l < j) quickSort(a, l, j);
    if (i < r) quickSort(a, i, r);
}
```

### 面试要点
- 平均复杂度 O(n log n)，最坏 O(n²)。
- 原地版本额外空间主要来自递归栈。
- 面试时必须解释边界推进，否则容易死循环。

## ALG-C-001 空一格环形缓冲区的push/pop与计数

```c
#define SIZE 8U

typedef struct {
    unsigned char buf[SIZE];
    size_t head;
    size_t tail;
} RingBuffer;

int ring_push(RingBuffer *rb, unsigned char value)
{
    if (rb == NULL)
        return -1;

    size_t next = (rb->head + 1U) % SIZE;
    if (next == rb->tail)
        return -1;

    rb->buf[rb->head] = value;
    rb->head = next;
    return 0;
}

int ring_pop(RingBuffer *rb, unsigned char *out)
{
    if (rb == NULL || out == NULL || rb->head == rb->tail)
        return -1;

    *out = rb->buf[rb->tail];
    rb->tail = (rb->tail + 1U) % SIZE;
    return 0;
}

size_t ring_count(const RingBuffer *rb)
{
    return (rb->head + SIZE - rb->tail) % SIZE;
}
```

空一格方案最大容量是`SIZE - 1`。索引推进必须环回；`rb->head++`最终可能越界。并发安全是另一个问题，不能由满空公式自动保证。

## ALG-C-002 带哨兵单链表的插入、删除与销毁

```c
typedef struct Node {
    int data;
    struct Node *next;
} Node;

int push_front(Node *head, int value)
{
    if (head == NULL)
        return -1;
    Node *node = malloc(sizeof *node);
    if (node == NULL)
        return -1;
    node->data = value;
    node->next = head->next;
    head->next = node;
    return 0;
}

int delete_value(Node *head, int value)
{
    if (head == NULL)
        return -1;

    Node *prev = head;
    while (prev->next != NULL) {
        if (prev->next->data == value) {
            Node *tmp = prev->next;
            prev->next = tmp->next;
            free(tmp);
            return 0;
        }
        prev = prev->next;
    }
    return -1;
}

void clear_list(Node *head)
{
    if (head == NULL)
        return;
    while (head->next != NULL) {
        Node *tmp = head->next;
        head->next = tmp->next;
        free(tmp);
    }
}
```

核心不变量：工作指针必须推进；删除时修改前驱的`next`，而不是只修改局部工作指针；失败返回应放在完整遍历结束后。

## ALG-C-003 带哨兵链式队列入队与出队

```c
typedef struct {
    Node *front;
    Node *rear;
} Queue;

int enqueue(Queue *q, int value)
{
    if (q == NULL || q->rear == NULL)
        return -1;

    Node *node = malloc(sizeof *node);
    if (node == NULL)
        return -1;
    node->data = value;
    node->next = NULL;

    q->rear->next = node;
    q->rear = node;
    return 0;
}

int dequeue(Queue *q, int *out)
{
    if (q == NULL || out == NULL || q->front == NULL ||
        q->front == q->rear)
        return -1;

    Node *tmp = q->front->next;
    *out = tmp->data;
    q->front->next = tmp->next;

    if (tmp == q->rear)
        q->rear = q->front;

    free(tmp);
    return 0;
}
```

出队删除的是哨兵后的首个有效节点，不是哨兵；通过`*out`修改调用者对象；删除最后一个有效节点后必须恢复`rear == front`这一空队列不变量。
